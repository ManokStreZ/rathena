prontera,147,173,5	script	Hunting Missions	4_CHN_GVG_01,{

	mes "[Hunting Missions]";
	mes "Halo, " + strcharinfo(0) + "!";
	mes "Poin Misi: ^0000FF" + #HuntingPoint + "^000000";
	next;

	// =============================
	// JIKA SEDANG ADA MISI AKTIF
	// =============================
	if (#MissionActive) {

		mes "[Status Misi]";
		mes "Target perburuanmu:";
		.@complete = 0;

		for (.@i = 0; .@i < #MissionCount; .@i++) {
			.@mid = getd("#Mission"+.@i);
			.@done = getd("#Mission"+.@i+"_");
			mes " > " + ((.@done >= #MissionAmount) ? "^00FF00" : "^FF0000")
			    + getmonsterinfo(.@mid, MOB_NAME)
			    + " ("+.@done+"/"+#MissionAmount+")^000000";

			if (.@done >= #MissionAmount)
				.@complete++;
		}
		next;

		switch(select("Lapor Selesai:Batalkan Misi:Tutup")) {

		case 1:
			if (.@complete < #MissionCount) {
				mes "[Hunting Missions]";
				mes "Target belum selesai!";
				close;
			}
			callsub L_Reward;
			close;

		case 2:
			callsub L_Reset;
			mes "[Hunting Missions]";
			mes "Misi dibatalkan.";
			close;

		case 3:
			close;
		}
	}

	// =============================
	// MENU UTAMA
	// =============================
	switch(select("Ambil Misi:Tukar Poin:Info:Keluar")) {

	case 1:
		if (#HuntingCooldown > gettimetick(2)) {
			mes "[Hunting Missions]";
			mes "Cooldown aktif!";
			mes "Sisa waktu:";
			mes "^FF0000" + Time2Str(#HuntingCooldown) + "^000000";
			close;
		}

		mes "[Hunting Missions]";
		mes "Pilih tingkat kesulitan:";
		next;

		.@diff = select("Easy:Normal:Hard (Mini Boss):Very Hard (MVP)");
		callsub L_Generate, .@diff;
		close;

	case 2:
		mes "[Hunting Missions]";
		mes "Poin kamu: ^0000FF" + #HuntingPoint + "^000000";
		close;

	case 3:
		mes "[Hunting Missions]";
		mes "Aturan:";
		mes "- 1 misi / 12 jam";
		mes "- Bunuh beberapa JENIS monster";
		mes "- Hadiah EXP + Poin";
		close;

	case 4:
		close;
	}

	end;

	// =============================
	// GENERATE MISSION
	// =============================
L_Generate:
	callsub L_Reset;

	#MissionDifficulty = getarg(0);

	if (#MissionDifficulty == 1) {
		#MissionCount = 10; #MissionAmount = 10;
		setarray .@mob[0], 1002,1001,1015,1016,1102,1004,1005,1007,1008,1011,1012,1013;
	}
	else if (#MissionDifficulty == 2) {
		#MissionCount = 20; #MissionAmount = 15;
		setarray .@mob[0], 1113,1111,1112,1115,1152,1119,1118,1117,1110,1109,1031,1032,1035;
	}
	else if (#MissionDifficulty == 3) {
		#MissionCount = 5; #MissionAmount = 1;
		setarray .@mob[0], 1096,1588,1120,1101,1190,1241,1272,1388,1089;
	}
	else if (#MissionDifficulty == 4) {
		#MissionCount = 3; #MissionAmount = 1;
		setarray .@mob[0], 1039,1038,1059,1041,1159,1147,1150,1115,1273;
	}

	for (.@i = 0; .@i < #MissionCount; .@i++) {

		.@pick = .@mob[rand(getarraysize(.@mob))];

		for (.@j = 0; .@j < .@i; .@j++) {
			if (getd("#Mission"+.@j) == .@pick) {
				.@i--;
				break;
			}
		}

		setd "#Mission"+.@i, .@pick;
		setd "#Mission"+.@i+"_", 0;
	}

	#MissionActive = 1;
	#HuntingCooldown = gettimetick(2) + 43200;

	mes "[Hunting Missions]";
	mes "Misi berhasil dibuat!";
	mes "Cek kembali NPC untuk melihat target.";
	return;

	// =============================
	// REWARD
	// =============================
L_Reward:
	if (#MissionDifficulty == 1) { .@p = 5;  .@exp = 200000; }
	else if (#MissionDifficulty == 2) { .@p = 10; .@exp = 600000; }
	else if (#MissionDifficulty == 3) { .@p = 15; .@exp = 2000000; }
	else if (#MissionDifficulty == 4) { .@p = 20; .@exp = 10000000; }

	#HuntingPoint += .@p;
	BaseExp += .@exp;
	JobExp  += .@exp;

	mes "[Hunting Missions]";
	mes "Hadiah diterima:";
	mes " > " + .@p + " Poin";
	mes " > " + F_InsertComma(.@exp) + " EXP";

	callsub L_Reset;
	specialeffect2 EF_ANGEL;
	return;

	// =============================
	// RESET
	// =============================
L_Reset:
	for (.@i = 0; .@i < 25; .@i++) {
		setd "#Mission"+.@i, 0;
		setd "#Mission"+.@i+"_", 0;
	}
	#MissionActive = 0;
	#MissionCount = 0;
	#MissionAmount = 0;
	#MissionDifficulty = 0;
	return;

	// =============================
	// KILL EVENT
	// =============================
OnNPCKillEvent:
	if (!#MissionActive) end;

	for (.@i = 0; .@i < #MissionCount; .@i++) {
		if (killedrid == getd("#Mission"+.@i)) {

			if (getd("#Mission"+.@i+"_") < #MissionAmount) {
				setd "#Mission"+.@i+"_", getd("#Mission"+.@i+"_") + 1;
				dispbottom "[Mission] " + getmonsterinfo(killedrid, MOB_NAME)
					+ " (" + getd("#Mission"+.@i+"_") + "/" + #MissionAmount + ")";
			}
			break;
		}
	}
	end;
}
